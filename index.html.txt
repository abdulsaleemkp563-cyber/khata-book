    <script>
        const DB_KEY = 'KHATA_ULTIMATE_V4_FINAL';
        let db = JSON.parse(localStorage.getItem(DB_KEY)) || { customers: [], trans: [], darkMode: true, pin: null, shopName: "MY BUSINESS", links: [] };
        let activeCid = null, pinInput = "";


        function checkAccess() {
            const id = new URLSearchParams(window.location.search).get('id');
            if (!id) { document.getElementById('accessDenied').style.display = 'flex'; return; }
            document.getElementById('lockScreen').style.display = 'flex';
            applyTheme();
        }


        function pressKey(n) { if (pinInput.length < 4) { pinInput += n; updateDots(); if (pinInput.length === 4) setTimeout(validate, 250); } }
        function updateDots() { document.querySelectorAll('.pin-dot').forEach((d, i) => d.classList.toggle('filled', i < pinInput.length)); }
        function clearPin() { pinInput = ""; updateDots(); }
        function validate() { if (!db.pin) { db.pin = pinInput; save(); unlock(); } else if (pinInput === db.pin) unlock(); else { alert("Wrong PIN!"); clearPin(); } }
        function unlock() { document.getElementById('lockScreen').style.display = 'none'; document.getElementById('mainApp').style.display = 'flex'; render(); }


        function applyTheme() { document.body.className = db.darkMode ? 'dark-mode' : 'light-mode'; document.getElementById('themeIcon').className = db.darkMode ? 'fa-solid fa-sun' : 'fa-solid fa-moon'; }
        function toggleTheme() { db.darkMode = !db.darkMode; applyTheme(); save(); }
        function save() { localStorage.setItem(DB_KEY, JSON.stringify(db)); render(); }


        function render() { activeCid ? renderDetails() : renderHome(); }


        function renderHome() {
            activeCid = null; document.getElementById('backBtn').classList.add('hidden'); document.getElementById('pageTitle').innerText = db.shopName;
            let gave = 0, got = 0; db.trans.forEach(t => { if(t.type === 'GAVE') gave += t.amt; else got += t.amt; });
            document.getElementById('mainContent').innerHTML = `
                <div class="dashboard-card mb-8"><p class="text-[10px] opacity-70 font-bold uppercase tracking-widest mb-3">Total Balance</p><h2 class="text-6xl font-black">₹${(gave - got).toLocaleString()}</h2><div class="flex justify-between mt-8 bg-black/10 p-5 rounded-3xl"><div><p class="text-[10px] opacity-50 font-bold">GAVE</p><p class="text-xl font-black text-red-300">₹${gave}</p></div><div class="w-[1px] bg-white/10"></div><div><p class="text-[10px] opacity-50 font-bold">GOT</p><p class="text-xl font-black text-green-300">₹${got}</p></div></div></div>
                <div class="mb-6 no-print"><input type="text" class="w-full p-5 rounded-3xl bg-gray-500/10 border-none outline-none font-semibold" placeholder="🔍 Search Customer..." oninput="renderCustList(this.value)"></div>
                <div id="custList" class="pb-32"></div>`;
            document.getElementById('bottomNav').innerHTML = `<button class="w-full bg-blue-600 py-5 rounded-3xl font-black text-white shadow-xl shadow-blue-500/30" onclick="openCustModal()"><i class="fa-solid fa-user-plus mr-3"></i> ADD CUSTOMER</button>`;
            renderCustList();
        }


        function renderCustList(q = "") {
            let filtered = db.customers.filter(c => c.name.toLowerCase().includes(q.toLowerCase()));
            document.getElementById('custList').innerHTML = filtered.map(c => {
                let bal = 0; db.trans.filter(t => t.cid === c.id).forEach(t => bal += (t.type === 'GAVE' ? t.amt : -t.amt));
                return `<div class="cust-card" onclick="view(${c.id})"><div class="w-14 h-14 bg-blue-600 rounded-2xl flex items-center justify-center font-black text-white mr-5 text-xl">${c.name[0]}</div><div class="flex-1 font-bold text-lg">${c.name}</div><div class="text-right font-black text-lg ${bal >= 0 ? 'text-red-500' : 'text-green-500'}">₹${Math.abs(bal)}</div></div>`;
            }).join('');
        }


        function view(id) { activeCid = id; render(); }
        function goHome() { activeCid = null; render(); }


        function renderDetails() {
            const cust = db.customers.find(c => c.id === activeCid);
            document.getElementById('backBtn').classList.remove('hidden'); document.getElementById('pageTitle').innerText = cust.name;
            let raw = db.trans.filter(t => t.cid === activeCid), bal = 0; raw.forEach(t => bal += (t.type === 'GAVE' ? t.amt : -t.amt));
            
            document.getElementById('mainContent').innerHTML = `
                <div class="print-head hidden"><h1>${db.shopName}</h1><p>Customer: ${cust.name}</p><h2>Balance: ₹${Math.abs(bal)}</h2></div>
                <div class="dashboard-card no-print mb-8"><h2 class="text-5xl font-black">₹${Math.abs(bal)}</h2><p class="text-[11px] opacity-70 uppercase mt-3 font-bold">${bal >= 0 ? 'You will give' : 'You will get'}</p><div class="flex justify-center gap-4 mt-8"><button class="bg-white/10 px-8 py-3 rounded-2xl text-[12px] font-black" onclick="openCustModal(${cust.id})">PROFILE</button><button class="bg-white text-blue-700 px-8 py-3 rounded-2xl text-[12px] font-black shadow-lg" onclick="window.print()">PRINT REPORT</button></div></div>
                <div class="print-section"><table class="w-full text-left"><thead><tr class="opacity-30 border-b text-[12px] uppercase font-black"><th>Date/Notes</th><th class="text-center">Gave</th><th class="text-center">Got</th><th class="text-right">Balance</th></tr></thead>
                <tbody>${raw.sort((a,b)=>new Date(b.date)-new Date(a.date)).map(t => `<tr class="border-b border-gray-500/5"><td class="py-5"><b>${t.date}</b><br><span class="opacity-50 text-xs">${t.remark || '-'}</span></td><td class="text-center text-red-500 font-black text-lg">${t.type==='GAVE'?'₹'+t.amt:'-'}</td><td class="text-center text-green-500 font-black text-lg">${t.type==='GOT'?'₹'+t.amt:'-'}</td><td class="text-right font-black text-lg opacity-80" onclick="openTransModal('${t.type}', ${t.id})">₹${t.amt}</td></tr>`).join('')}</tbody></table></div><div class="h-40"></div>`;
            
            document.getElementById('bottomNav').innerHTML = `<div class="flex gap-4"><button class="flex-1 bg-red-500 py-5 rounded-3xl font-black text-white text-lg shadow-xl shadow-red-500/20" onclick="openTransModal('GAVE')">GAVE ₹</button><button class="flex-1 bg-green-600 py-5 rounded-3xl font-black text-white text-lg shadow-xl shadow-green-600/20" onclick="openTransModal('GOT')">GOT ₹</button></div>`;
        }


        function openModal(h) { document.getElementById('modalBox').innerHTML = h; document.getElementById('modalOverlay').style.display = 'flex'; setTimeout(()=>document.getElementById('modalBox').classList.add('active'), 10); }
        function closeModal() { document.getElementById('modalBox').classList.remove('active'); setTimeout(()=>document.getElementById('modalOverlay').style.display='none', 300); }
        
        function openCustModal(id=null) {
            const c = id ? db.customers.find(x=>x.id===id) : {name:'', phone:''};
            openModal(`<h3 class="text-xl font-black mb-8 uppercase">Customer Profile</h3><input type="text" id="cName" class="w-full p-5 rounded-2xl bg-gray-500/10 mb-4 outline-none font-bold" placeholder="Name" value="${c.name}"><input type="number" id="cPhone" class="w-full p-5 rounded-2xl bg-gray-500/10 mb-8 outline-none font-bold" placeholder="WhatsApp No" value="${c.phone}"><button class="w-full bg-blue-600 py-5 rounded-2xl font-black text-white shadow-lg" onclick="saveCust(${id})">SAVE CUSTOMER</button>${id ? `<button class="w-full mt-6 text-red-500 font-bold" onclick="deleteCust(${id})">DELETE ACCOUNT</button>` : ''}`);
        }
        function saveCust(id) { let n = document.getElementById('cName').value; if(!n) return; if(id) { let x=db.customers.find(y=>y.id===id); x.name=n; x.phone=document.getElementById('cPhone').value; } else { db.customers.push({id:Date.now(), name:n, phone:document.getElementById('cPhone').value}); } save(); closeModal(); }
        function deleteCust(id) { if(confirm("Account ഒഴിവാക്കണോ?")) { db.customers=db.customers.filter(x=>x.id!==id); db.trans=db.trans.filter(x=>x.cid!==id); save(); closeModal(); goHome(); } }


        function openTransModal(type, editId=null) {
            const t = editId ? db.trans.find(x=>x.id===editId) : {amt:'', remark:'', date:new Date().toISOString().split('T')[0]};
            openModal(`<h3 class="text-xl font-black mb-8 uppercase text-center">${type} Entry</h3><input type="number" id="tAmt" class="w-full p-5 rounded-2xl bg-gray-500/10 mb-4 text-4xl font-black text-center text-blue-500" value="${t.amt}" placeholder="₹ 0"><input type="text" id="tRem" class="w-full p-5 rounded-2xl bg-gray-500/10 mb-4 outline-none font-bold" value="${t.remark}" placeholder="Notes"><input type="date" id="tDate" class="w-full p-5 rounded-2xl bg-gray-500/10 mb-8 outline-none font-bold" value="${t.date}"><button class="w-full bg-blue-600 py-5 rounded-2xl font-black text-white shadow-lg" onclick="saveTrans('${type}', ${editId})">CONFIRM & SAVE</button>${editId ? `<button class="w-full mt-6 text-red-500 font-bold" onclick="deleteTrans(${editId})">DELETE ENTRY</button>` : ''}`);
        }
        function saveTrans(type, id) {
            let a = parseFloat(document.getElementById('tAmt').value); if(!a) return;
            if(id) { let x=db.trans.find(y=>y.id===id); x.amt=a; x.date=document.getElementById('tDate').value; x.remark=document.getElementById('tRem').value; }
            else { db.trans.push({id:Date.now(), cid:activeCid, type, amt:a, date:document.getElementById('tDate').value, remark:document.getElementById('tRem').value}); }
            save(); closeModal();
            const cust = db.customers.find(c => c.id === activeCid);
            if(cust.phone && confirm("WhatsApp message അയക്കണോ?")) {
                let bal = 0; db.trans.filter(t => t.cid === activeCid).forEach(t => bal += (t.type === 'GAVE' ? t.amt : -t.amt));
                const msg = `🧾 *Entry from ${db.shopName}*%0A---------------------------%0A*Amount:* ₹${a}%0A*Status:* ${type}%0A*Balance:* ₹${Math.abs(bal)}%0A---------------------------`;
                window.open(`https://wa.me/91${cust.phone}?text=${msg}`, '_blank');
            }
        }
        function deleteTrans(id) { if(confirm("Delete?")) { db.trans=db.trans.filter(x=>x.id!==id); save(); closeModal(); } }


        function openAdmin() {
            let linksHtml = (db.links || []).map(l => `<div class="flex justify-between items-center p-4 bg-blue-500/5 rounded-2xl mb-3"><span class="font-bold text-xs uppercase">${l.name}</span><button onclick="copyLink('${l.url}')" class="text-blue-500 font-black text-[10px] bg-blue-500/10 px-4 py-2 rounded-lg">COPY</button></div>`).join('');
            openModal(`<h3 class="text-xl font-black mb-8">ADMIN PANEL</h3>
                <input type="text" id="sName" class="w-full p-4 rounded-xl bg-gray-500/10 mb-4 font-bold" value="${db.shopName}" placeholder="Shop Name">
                <button class="w-full bg-blue-600 py-3 rounded-xl font-black text-white mb-8" onclick="db.shopName=document.getElementById('sName').value; save(); closeModal();">UPDATE NAME</button>
                <div class="grid grid-cols-2 gap-4 mb-8">
                    <button class="bg-emerald-600 py-4 rounded-xl font-black text-[10px] text-white" onclick="exportJSON()">BACKUP DATA</button>
                    <button class="bg-amber-600 py-4 rounded-xl font-black text-[10px] text-white" onclick="document.getElementById('importFile').click()">RESTORE DATA</button>
                    <input type="file" id="importFile" class="hidden" accept=".json" onchange="importJSON(event)">
                </div>
                <div class="flex gap-2 mb-4"><input type="text" id="newUid" class="flex-1 p-4 rounded-xl bg-gray-500/10 text-xs font-bold" placeholder="Staff Name"><button class="bg-blue-600 px-6 rounded-xl font-black text-white text-[10px]" onclick="genLink()">GET LINK</button></div>
                <div class="max-h-32 overflow-y-auto mb-4">${linksHtml}</div>
                <button class="w-full text-red-500 font-bold text-xs mt-4" onclick="if(confirm('Reset PIN?')) {db.pin=null;save();location.reload();}">RESET SECURITY PIN</button>`);
        }
        function genLink() { let n = document.getElementById('newUid').value.trim(); if(!n) return; let url = window.location.href.split('?')[0] + '?id=' + n.toLowerCase(); db.links.push({name:n, url}); save(); openAdmin(); }
        function copyLink(u) { navigator.clipboard.writeText(u).then(()=>alert("Copied!")); }
        
        function exportJSON() { 
            const b = new Blob([JSON.stringify(db)], {type:'application/json'}); 
            const a = document.createElement('a'); 
            a.href=URL.createObjectURL(b); 
            a.download=`Khata_Backup_${new Date().toLocaleDateString()}.json`; 
            a.click(); 
        }


        function importJSON(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const importedDb = JSON.parse(e.target.result);
                    if (confirm("ഈ ഫയലിലെ ഡാറ്റ ഇമ്പോർട്ട് ചെയ്യണോ? നിലവിലുള്ള കണക്കുകൾ മാഞ്ഞുപോകും!")) {
                        db = importedDb;
                        save();
                        location.reload();
                    }
                } catch (err) { alert("തെറ്റായ ഫയൽ ആണ് നിങ്ങൾ തിരഞ്ഞെടുത്തത്!"); }
            };
            reader.readAsText(file);
        }
